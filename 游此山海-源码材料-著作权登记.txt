    {
        public int Seed { get; private set; }
        public Dictionary<string, Region> Regions = new Dictionary<string, Region>();
        public List<string> discovered = new List<string>();

        public ShanHaiWorld(int seed)
        {
            Seed = seed;
            GenerateRegions();
        }

        private void GenerateRegions()
        {
            // 按照山海经的方位大致分的，后面会再调
            AddRegion("昆仑丘", RegionType.Sacred, 0.3f);
            AddRegion("青丘", RegionType.FoxLand, 0.45f);
            AddRegion("流波山", RegionType.Wild, 0.6f);
            AddRegion("不周山", RegionType.Ruins, 0.85f);
            AddRegion("大泽", RegionType.Water, 0.55f);
            AddRegion("长留山", RegionType.Spirit, 0.4f);
            // 后面还有好多，先弄这几个能跑就行
        }

        void AddRegion(string name, RegionType t, float danger)
        {
            var r = new Region(name, t, danger, Seed);
            Regions[name] = r;
        }

        public Region GetRegion(string name)
        {
            if (Regions.ContainsKey(name))
                return Regions[name];
            Debug.LogWarning("找不到区域: " + name);
            return null;
        }
    }

    public enum RegionType
    {
        Sacred,
        FoxLand,
        Wild,
        Ruins,
        Water,
        Spirit,
        Human
    }

// 2026 年 1 月
import express from 'express';
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';
import crypto from 'crypto';
const __dirname = path.dirname(fileURLToPath(import.meta.url));
const DATA_DIR = path.join(__dirname, '..', 'server-data');
const USERS = path.join(DATA_DIR, 'users.json');
const SAVES = path.join(DATA_DIR, 'saves');
if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });
if (!fs.existsSync(SAVES)) fs.mkdirSync(SAVES, { recursive: true });
if (!fs.existsSync(USERS)) fs.writeFileSync(USERS, '[]');
const app = express();
app.use(express.json({ limit: '2mb' }));

function hash(pwd) {
  return crypto.createHash('sha256').update(pwd).digest('hex');
}
app.post('/api/register', (req, res) => {
  const { account, pwd } = req.body;
  if (!account || !pwd) return res.status(400).json({ err: '参数不全' });
  const users = JSON.parse(fs.readFileSync(USERS, 'utf8'));
  if (users.find(u => u.account === account)) {
    return res.status(409).json({ err: '账号已存在' });
  }
  users.push({ account, pwd: hash(pwd), created: Date.now() });
  fs.writeFileSync(USERS, JSON.stringify(users, null, 2));
  res.json({ ok: true });
});
app.post('/api/login', (req, res) => {
  const { account, pwd } = req.body;
  const users = JSON.parse(fs.readFileSync(USERS, 'utf8'));
  const u = users.find(x => x.account === account && x.pwd === hash(pwd));
  if (!u) return res.status(401).json({ err: '账号或密码错误' });
  res.json({ ok: true, token: account + ':' + Date.now() });
});
app.post('/api/save', (req, res) => {
  const { account, data } = req.body;
  if (!account) return res.status(400).json({ err: '缺少账号' });
  fs.writeFileSync(path.join(SAVES, account + '.json'), JSON.stringify(data, null, 2));
  res.json({ ok: true });
});
app.get('/api/load/:account', (req, res) => {
  const f = path.join(SAVES, req.params.account + '.json');
  if (!fs.existsSync(f)) return res.json({ ok: true, data: null });
  const raw = fs.readFileSync(f, 'utf8');
  res.json({ ok: true, data: JSON.parse(raw) });
});
const PORT = process.env.PORT || 8787;
app.listen(PORT, () => {
  console.log('游此山海云存档服务已启动 :', PORT);
});

// App.tsx
// 这是主文件
// 2026-02 最后一次大改，把钱袋弹窗和核心玩法说明放一起了
import React, { useState } from 'react';
import { useGameStore } from './store/gameStore';
import type { Scene } from './types';
import { STONE_BASE, STONE_PER_LV, TRIP_BASE_MIN, SEC_PER_GRADE, SEC_PER_TOOL } from './constants';
// 下面这些 import 后面再整理
import CourtScene from './components/CourtScene';
import MarketScene from './components/MarketScene';
import BagScene from './components/BagScene';
import AtlasScene from './components/AtlasScene';
import DiaryScene from './components/DiaryScene';
import MoneyModal from './components/MoneyModal';
export default function App() {
  const store = useGameStore();
  const [scene, setScene] = useState<Scene>('court');
  const [showMoney, setShowMoney] = useState(false);
  // 简单判断能不能出发
  const canDepart = React.useMemo(() => {
    const hasFood = store.bag.some(b => b.kind === 'food' && b.count > 0);
    const hasHerb = store.bag.some(b => b.kind === 'herb' && b.count > 0);
    const cost = STONE_BASE + (store.kid.lv - 1) * STONE_PER_LV;
    return hasFood && hasHerb && store.stones >= cost;
  }, [store.bag, store.kid.lv, store.stones]);
  // 计算本次出发大概要多久（秒）
  function calcTripSeconds(): number {
    let sec = TRIP_BASE_MIN * 60;
    // 吃食档位
    const foods = store.bag.filter(b => b.kind === 'food');
    for (const f of foods) sec -= f.grade * SEC_PER_GRADE;
    // 灵药档位
    const herbs = store.bag.filter(b => b.kind === 'herb');
    for (const h of herbs) sec -= h.grade * SEC_PER_GRADE;
    // 道具
    const tools = store.bag.filter(b => b.kind === 'tool');
    for (const t of tools) sec -= t.grade * SEC_PER_TOOL;

    const minSec = TRIP_BASE_MIN * 60 * 0.2; // 最少剩 20%
    return Math.max(Math.floor(sec), minSec);
  }
  // 点击出发（真正跑逻辑在 CourtScene 里）
  function handleDepart() {
    if (!canDepart) return;
    // 这里只是把场景切走，实际倒计时在 CourtScene
    setScene('court');
  }
  return (
    <div className="min-h-screen bg-[#f5f0e6] text-[#3d2b1f] pb-16">
      {/* 顶部状态条 */}
      <div className="sticky top-0 z-50 bg-[#f5f0e6]/95 border-b border-[#d4c9b5] px-4 py-2 flex items-center justify-between">
        <div className="flex items-center gap-2">
          <div className="text-xl font-bold tracking-wider">游此山海</div>
          <div className="text-xs text-[#8a7259]">v1.3.2</div>
        </div>
        <button
          onClick={() => setShowMoney(true)}
          className="flex items-center gap-1 px-3 py-1 rounded-full bg-[#5c4033] text-white text-sm active:opacity-80"
        >
          <span>钱袋</span>
          <span className="font-mono">{store.stones}</span>
        </button>
      </div>
      {/* 主场景 */}
      <div className="p-4">
        {scene === 'court' && (
          <CourtScene
            store={store}
            canDepart={canDepart}
            onDepart={handleDepart}
            onOpenMoney={() => setShowMoney(true)}
          />
        )}
        {scene === 'market' && <MarketScene store={store} />}
        {scene === 'bag' && <BagScene store={store} />}
        {scene === 'atlas' && <AtlasScene store={store} />}
        {scene === 'diary' && <DiaryScene store={store} />}
      </div>
      {/* 底部导航 */}
      <div className="fixed bottom-0 left-0 right-0 max-w-[480px] mx-auto bg-white border-t border-[#d4c9b5] flex">
        {[
          { key: 'court', label: '庭院' },
          { key: 'market', label: '市集' },
          { key: 'bag', label: '行囊' },
          { key: 'atlas', label: '图鉴' },
          { key: 'diary', label: '日记' },
        ].map(item => (
          <button
            key={item.key}
            onClick={() => setScene(item.key as Scene)}
            className={`flex-1 py-3 text-sm ${scene === item.key ? 'nav-active' : 'text-[#8a7259]'}`}
          >
            {item.label}
          </button>
        ))}
      </div>
      {/* 钱袋弹窗（核心玩法说明在这里） */}
      <MoneyModal
        open={showMoney}
        onClose={() => setShowMoney(false)}
        stones={store.stones}
        kidLv={store.kid.lv}
      />
    </div>
  );
}

#!/usr/bin/env node
/#
 /#
 从「图鉴照片」文件夹按文件名前的序号排序，复制到 public/herbalist-trip/，
 */#
并生成 src/herbalistTripPhotos.manifest.json 供应用引用。
 //用法：解决
 //  node scripts/sync-herbalist-trip-photos.mjs
 //SOURCE_DIR="/path/to/图鉴照片" node scripts/sync-herbalist-trip-photos.mjs
 import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
const __dirname = path.dirname(fileURLToPath(import.meta.url));
const root = path.resolve(__dirname, '..');
const destDir = path.join(root, 'public', 'herbalist-trip');
const manifestPath = path.join(root, 'src', 'herbalistTripPhotos.manifest.json');
const defaultSource = '/Users/guocong/Desktop/图鉴照片';
const sourceDir = process.env.SOURCE_DIR || defaultSource;
const IMAGE_EXT = new Set(['.jpg', '.jpeg', '.png', '.webp', '.gif']);
function leadingInt(basename) {
  const m = basename.match(/^\s*(\d+)/);
  return m ? parseInt(m[1], 10) : Number.POSITIVE_INFINITY;
}
function main() {
  if (!fs.existsSync(sourceDir)) {
    console.error(`源目录不存在: ${sourceDir}`);
    console.error('可设置环境变量 SOURCE_DIR 指向你的「图鉴照片」文件夹。');
    process.exit(1);
  }
  fs.mkdirSync(destDir, { recursive: true });
  for (const name of fs.readdirSync(destDir)) {
    if (name === '.DS_Store' || name === 'manifest.json') continue;
    fs.unlinkSync(path.join(destDir, name));
  }
  const entries = fs
    .readdirSync(sourceDir)
    .filter((name) => !name.startsWith('.') && name !== '.DS_Store')
    .map((name) => {
      const full = path.join(sourceDir, name);
      const st = fs.statSync(full);
      if (!st.isFile()) return null;
      const ext = path.extname(name).toLowerCase();
      if (!IMAGE_EXT.has(ext)) return null;
      const key = leadingInt(name);
      return { name, full, ext, key };
    })
    .filter(Boolean);

  entries.sort((a, b) => {
    if (a.key !== b.key) return a.key - b.key;
    return a.name.localeCompare(b.name, 'zh-Hans-CN');
  });
  const relPaths = [];
  entries.forEach((e, i) => {
    const destName = `${String(i + 1).padStart(3, '0')}${e.ext}`;
    fs.copyFileSync(e.full, path.join(destDir, destName));
    relPaths.push(`herbalist-trip/${destName}`);
  });
  fs.writeFileSync(manifestPath, `${JSON.stringify(relPaths, null, 2)}\n`, 'utf8');
  console.log(`已复制 ${relPaths.length} 张图 -> public/herbalist-trip/`);
  console.log(`已写入 ${path.relative(root, manifestPath)}`);
}
main();

  const [countdown, setCountdown] = useState(0);
  const [isTripping, setIsTripping] = useState(false);
  // 简单模拟倒计时（真实项目里用 requestAnimationFrame 更准）
  useEffect(() => {
    if (!isTripping || countdown <= 0) return;
    const t = setTimeout(() => {
      const next = countdown - 1;
      setCountdown(next);
      if (next <= 0) {
        // 游历结束
        finishTrip();
      }
    }, 1000);
    return () => clearTimeout(t);
  }, [countdown, isTripping]);
  function startTrip() {
    if (!canDepart) return;
    // 扣东西
    const cost = STONE_BASE + (store.kid.lv - 1) * STONE_PER_LV;
    if (!store.spendStones(cost)) return;

    const food = store.bag.find(b => b.kind === 'food' && b.count > 0);
    const herb = store.bag.find(b => b.kind === 'herb' && b.count > 0);
    if (food) store.removeFromBag(food.id, 1);
    if (herb) store.removeFromBag(herb.id, 1);
    // 开始倒计时（这里简化成 90 秒演示）
    setCountdown(90);
    setIsTripping(true);
  }
  function finishTrip() {
    setIsTripping(false);
    setCountdown(0);
    // 随机给点奖励
    const newIds: number[] = [];
    store.atlas.forEach(a => {
      if (!a.unlocked && Math.random() > 0.6) newIds.push(a.id);
    });
    if (newIds.length) store.unlockAtlas(newIds);
    // 加经验
    const exp = 40 + Math.floor(Math.random() * 30);
    store.gainExp(exp);
    // TODO: 这里应该弹出结果弹窗，后面再说
    alert(`游历结束！获得 ${exp} 经验${newIds.length ? `，新图鉴 ${newIds.length} 个` : ''}`);
  }
  return (
    <div className="space-y-4">
      {/* 天候提示 */}
      <div className="paper-panel p-4 text-sm">
        <div className="flex justify-between items-start">
          <div>
            <div className="text-[#8a7259] text-xs">天候</div>
            <div className="text-lg font-medium">晴光好</div>

import { useState, useEffect } from 'react';
export function useTripTimer(initialSeconds: number) {
  const [sec, setSec] = useState(initialSeconds);
  const [running, setRunning] = useState(false);
  useEffect(() => {
    if (!running || sec <= 0) return;
    const t = setTimeout(() => setSec(s => s - 1), 1000);
    return () => clearTimeout(t);
  }, [running, sec]);
  return {
    seconds: sec,
    start: () => setRunning(true),
    stop: () => setRunning(false),
    reset: (n: number) => { setSec(n); setRunning(false); },
    isRunning: running,
  };
}

// 自己写的简单状态管理，没用 redux / zustand
// 后面觉得太麻烦就直接用 localStorage + useState 了
import { useState, useEffect } from 'react';
import type { KidState, BagItem, AtlasEntry, DiaryEntry, MoneyBag } from '../types';
import { INIT_KID, INIT_STONES, ATLAS_SEED } from '../constants';
const LS_KEY = 'ycsh_save_v3';
interface SaveData {
  kid: KidState;
  stones: number;
  bag: BagItem[];
  atlas: AtlasEntry[];
  diary: DiaryEntry[];
  version: number;
}
function load(): SaveData {
  try {
    const raw = localStorage.getItem(LS_KEY);
    if (!raw) throw new Error('no save');
    const data = JSON.parse(raw) as SaveData;
    // 版本兼容处理，旧存档可能缺字段
    if (!data.atlas) data.atlas = [...ATLAS_SEED];
    if (!data.diary) data.diary = [];
    return data;
  } catch {
    // 新号给初始数据
    return {
      kid: { ...INIT_KID },
      stones: INIT_STONES,
      bag: [],
      atlas: [...ATLAS_SEED],
      diary: [],
      version: 3,
    };
  }
}
function save(data: SaveData) {
  localStorage.setItem(LS_KEY, JSON.stringify(data));
}
export function useGameStore() {
  const [data, setData] = useState<SaveData>(() => load());
  // 自动保存
  useEffect(() => {
    save(data);
  }, [data]);
  const kid = data.kid;
  const stones = data.stones;
  const bag = data.bag;
  const atlas = data.atlas;
  const diary = data.diary;
  // 下面这些函数
  function addStones(n: number) {
    setData(d => ({ ...d, stones: Math.max(0, d.stones + n) }));
  }
  function spendStones(n: number): boolean {
    if (data.stones < n) return false;
    setData(d => ({ ...d, stones: d.stones - n }));
    return true;
  }
  function addToBag(item: BagItem) {
    setData(d => {
      const exist = d.bag.findIndex(b => b.id === item.id);
      if (exist >= 0) {
        const copy = [...d.bag];
        copy[exist] = { ...copy[exist], count: copy[exist].count + item.count };
        return { ...d, bag: copy };
      }
      return { ...d, bag: [...d.bag, item] };
    });
  }
  function removeFromBag(id: string, count: number) {
    setData(d => {
      const idx = d.bag.findIndex(b => b.id === id);
      if (idx < 0) return d;
      const copy = [...d.bag];
      const now = copy[idx];
      if (now.count <= count) {
        copy.splice(idx, 1);
      } else {
        copy[idx] = { ...now, count: now.count - count };
      }
      return { ...d, bag: copy };
    });
  }
  function unlockAtlas(ids: number[]) {
    setData(d => ({
      ...d,
      atlas: d.atlas.map(a =>
        ids.includes(a.id) ? { ...a, unlocked: true } : a
      ),
    }));
  }
  function addDiary(entry: DiaryEntry) {
    setData(d => ({ ...d, diary: [...d.diary, entry] }));
  }
  function gainExp(n: number) {
    setData(d => {
      let { lv, exp, next } = d.kid;
      exp += n;
      while (exp >= next) {
        exp -= next;
        lv += 1;
        next = Math.floor(next * 1.6);
      }
      return {
        ...d,
        kid: { ...d.kid, lv, exp, next },
      };
    });
  }
  function resetAll() {
    // 调试用，正式版可以删
    localStorage.removeItem(LS_KEY);
    window.location.reload();
  }
  return {
    kid,
    stones,
    bag,
    atlas,
    diary,
    addStones,
    spendStones,
    addToBag,
    removeFromBag,
    unlockAtlas,
    addDiary,
    gainExp,
    resetAll,
  };
}

// utils/calc.ts
// 游历时间计算逻辑，单独抽出来方便后面改
import { SEC_PER_GRADE, SEC_PER_TOOL, TRIP_BASE_MIN } from '../constants';
import type { BagItem } from '../types';
export function calcTripSeconds(bag: BagItem[]): number {
  let sec = TRIP_BASE_MIN * 60;
  const foods = bag.filter(b => b.kind === 'food');
  for (const f of foods) {
    sec -= f.grade * SEC_PER_GRADE;
  }
  const herbs = bag.filter(b => b.kind === 'herb');
  for (const h of herbs) {
    sec -= h.grade * SEC_PER_GRADE;
  }
  const tools = bag.filter(b => b.kind === 'tool');
  for (const t of tools) {
    sec -= t.grade * SEC_PER_TOOL;
  }
  // 最少剩 20%
  const min = Math.floor(TRIP_BASE_MIN * 60 * 0.2);
  return Math.max(sec, min);
}
